Djangoモデル（models.py イメージ）
from django.db import models
from django.contrib.auth.models import AbstractUser

class User(AbstractUser):
    department = models.CharField(max_length=100)
    is_manager = models.BooleanField(default=False)

class DepartmentGoal(models.Model):
    department = models.CharField(max_length=100)
    year = models.IntegerField()
    philosophy = models.TextField()
    finance = models.TextField()
    safety = models.TextField()
    cooperation = models.TextField()

class PersonalGoal(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    year = models.IntegerField()
    philosophy_goal = models.TextField()
    finance_goal = models.TextField()
    safety_goal = models.TextField()
    cooperation_goal = models.TextField()

class Achievement(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    year = models.IntegerField()
    philosophy_result = models.TextField()
    finance_result = models.TextField()
    safety_result = models.TextField()
    cooperation_result = models.TextField()

class Evaluation(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    year = models.IntegerField()
    philosophy_eval = models.TextField()
    finance_eval = models.TextField()
    safety_eval = models.TextField()
    cooperation_eval = models.TextField()
CSV読み込み処理（初回起動時）
# management/commands/import_users.py

import csv
from django.core.management.base import BaseCommand
from app.models import User

class Command(BaseCommand):
    def handle(self, *args, **kwargs):
        with open('users.csv', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                user, created = User.objects.get_or_create(
                    username=row['id'],
                    defaults={
                        'first_name': row['名前'],
                        'department': row['部署'],
                        'is_manager': bool(int(row['上長フラグ']))
                    }
                )
                user.set_password(row['パスワード'])
                user.save()
